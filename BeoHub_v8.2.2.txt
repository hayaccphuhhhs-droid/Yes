local function makeDrag(frame, handle)
    local dragging, startMouse, startPos
    handle.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            startMouse = i.Position
            startPos = frame.Position
            i.Changed:Connect(function()
                if i.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    uis.InputChanged:Connect(function(i)
        if not dragging then return end
        if i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch then
            local d = i.Position - startMouse
            frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end)
end

    hit.Size = UDim2.new(1, 0, 3, 0)
    hit.Position = UDim2.new(0, 0, -1, 0)
    hit.BackgroundTransparency = 1
    hit.Text = ""
    hit.MouseButton1Down:Connect(function()
        local dragging = true
        local c1, c2
        c1 = uis.InputChanged:Connect(function(i)
            if not dragging then return end
            if i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch then
                local rel = math.clamp((i.Position.X - bar.AbsolutePosition.X) / math.max(bar.AbsoluteSize.X, 1), 0, 1)
                local step = (max - min) <= 5 and 0.05 or 1
                local val = min + (max - min) * rel
                if step >= 1 then val = math.floor(val + 0.5)
                else val = math.floor(val / step + 0.5) * step; val = math.floor(val * 100 + 0.5) / 100 end
                fill.Size = UDim2.new(rel, 0, 1, 0)
                l.Text = t .. " : " .. tostring(val)
                set(val)
                saveCfg()
            end
        end)
        c2 = uis.InputEnded:Connect(function(i)
            if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
                dragging = false
                if c1 then c1:Disconnect() end
                if c2 then c2:Disconnect() end
            end
        end)
    end)
end

local function getBuyableBaitList()
    local list, seen = {}, {}
    local info = rep:FindFirstChild("Info")
    local baitFolder = info and info:FindFirstChild("Bait")
    if baitFolder then
        for _, c in ipairs(baitFolder:GetChildren()) do
            if c:IsA("ModuleScript") then
                local ok, data = pcall(require, c)
                if ok and type(data) == "table" and not data.Ingredient and not seen[c.Name] then
                    seen[c.Name] = true
                    table.insert(list, c.Name)
                end
            end
        end
    end
    local itemFolder = info and info:FindFirstChild("Item")
    if itemFolder then
        for _, c in ipairs(itemFolder:GetChildren()) do
            if c:IsA("ModuleScript") and not seen[c.Name] then
                local ok, data = pcall(require, c)
                if ok and type(data) == "table" and data.Type == "Bait" and not data.Ingredient then
                    seen[c.Name] = true
                    table.insert(list, c.Name)
                end
            end
        end
    end
    table.sort(list)
    return list
end

function setWalkOnWater(on)
    if on then
        for _, v in pairs(workspace:GetDescendants()) do
            if v:IsA("BasePart") and v.Name:lower() == "water" then
                if oW[v] == nil then oW[v] = v.CanCollide end
                v.CanCollide = true
            end
        end
        local ocean = workspace:FindFirstChild("Ocean")
        if ocean then
            for _, waterPart in ipairs(ocean:GetChildren()) do
                if waterPart.Name == "Water" and waterPart:IsA("BasePart") then
                    if oW[waterPart] == nil then oW[waterPart] = waterPart.CanCollide end

end

local lastDailyQuestClock, lastDailyRewardClock = 0, 0
local function stepAutoClaimDailyQuest()
    if not getgenv().AutoClaimDaily or getgenv().AdminIn then return end
    local now = os.clock()
    if now - lastDailyQuestClock < 2 then return end
    lastDailyQuestClock = now
    pcall(function()
        local pData = getPlayerData()
        local questFolder = pData and pData:FindFirstChild("Quest")
        local dailyFolder = questFolder and questFolder:FindFirstChild("Daily")
        if not dailyFolder then return end
        local questInfoTable = nil
        local questInfoMod = InfoFolder and InfoFolder:FindFirstChild("Quest")
        if questInfoMod then pcall(function() questInfoTable = require(questInfoMod) end) end
        local claimEv = evs:FindFirstChild("ClaimQuest")
        if not claimEv then return end
        for _, dailyEntry in ipairs(dailyFolder:GetChildren()) do
            local claimVal = dailyEntry:FindFirstChild("Claim")
            local progVal = dailyEntry:FindFirstChild("Progression")
            local questTypeVal = dailyEntry:FindFirstChild("QuestType")
            if claimVal and progVal and questTypeVal and claimVal.Value == false and questTypeVal.Value ~= "" then
                local reqAmount = nil
                if questInfoTable and questInfoTable.Daily and tonumber(dailyEntry.Name) then
                    local dailyDef = questInfoTable.Daily[tonumber(dailyEntry.Name)]
                    if dailyDef and dailyDef[2] then reqAmount = dailyDef[2] end
                end
                if reqAmount then
                    local curProg = progVal.Value
                    if questTypeVal.Value == "PlayTime" then
                        if math.floor(curProg / 60) >= math.floor(reqAmount / 60) then
                            claimEv:FireServer(dailyEntry.Name)
                        end
                    else
                        if curProg >= reqAmount then
                            claimEv:FireServer(dailyEntry.Name)
                        end
                    end
                end
            end
        end
    end)
end

local function cleanBaitName(s)
    return tostring(s or ""):gsub("^%s+", ""):gsub("%s+$", "")
end

local function getBaitItemCount(inst)
    if not inst then return 0 end
    local n = 0
    pcall(function()
        if inst:IsA("NumberValue") or inst:IsA("IntValue") then
            n = tonumber(inst.Value) or 0
        else
            local q = inst:FindFirstChild("Quantity") or inst:FindFirstChild("Amount") or inst:FindFirstChild("Count")
            if q and q:IsA("ValueBase") then
                n = tonumber(q.Value) or 0
            else
                n = 1
            end
        end
    end)
    return math.max(0, n)
end

local function getBasicBaitCount()
    local pData = getPlayerData()
    if not pData then return 0 end
    local total = 0
    local function scan(folder)
        if not folder then return end
        for _, child in ipairs(folder:GetChildren()) do
            local n = cleanBaitName(child.Name):lower()
            if n == "basic bait" then
                total = total + getBaitItemCount(child)
            end
        end
    end
    scan(pData:FindFirstChild("Bait"))
    scan(pData:FindFirstChild("Inventory"))
    return total
end

local function getEquippedBaitQuantity()
    local equippedQty = -1
    pcall(function()
        local mainGui = me:FindFirstChild("PlayerGui") and me.PlayerGui:FindFirstChild("MainGui")
        local hotbarList = mainGui
            and mainGui:FindFirstChild("Main")
            and mainGui.Main:FindFirstChild("Hotbar")
            and mainGui.Main.Hotbar:FindFirstChild("List")
        local slot2Frame = hotbarList and hotbarList:FindFirstChild("2")
        if slot2Frame then
            local qtyLabel = slot2Frame:FindFirstChild("Quantity")
                or (slot2Frame:FindFirstChild("Quantity_BG") and slot2Frame.Quantity_BG:FindFirstChild("Quantity"))
                or (slot2Frame:FindFirstChild("Detail") and slot2Frame.Detail:FindFirstChild("Quantity"))
                or (slot2Frame:FindFirstChild("Inner") and slot2Frame.Inner:FindFirstChild("Quantity"))
            if qtyLabel and qtyLabel:IsA("TextLabel") then
                if not qtyLabel.Visible or qtyLabel.Text == "" then
                    equippedQty = 0
                else
                    equippedQty = tonumber(string.match(qtyLabel.Text, "%d+")) or 0
                end
            end
        end
    end)
    if equippedQty == -1 then
        local pData = getPlayerData()
        pcall(function()
            local slot2 = pData and pData:FindFirstChild("Hotbar") and pData.Hotbar:FindFirstChild("2")
            if slot2 then
                local q = slot2:FindFirstChild("Quantity")
                if q and q:IsA("ValueBase") then
                    equippedQty = tonumber(q.Value) or 0
                end
            end
        end)
    end
    return math.max(0, equippedQty)
end

local function getEquippedBaitName()
    local equippedName = nil
    local pData = getPlayerData()
    pcall(function()
        if pData then
            local slot2 = pData:FindFirstChild("Hotbar") and pData.Hotbar:FindFirstChild("2")
            if slot2 then
                local vn = slot2:FindFirstChild("ValueName")
                local q = slot2:FindFirstChild("Quantity")
                local qty = q and tonumber(q.Value) or 0
                if vn and tostring(vn.Value) ~= "" and qty > 0 then
                    equippedName = cleanBaitName(vn.Value)
                end
            end
            if not equippedName then
                local eq = pData:FindFirstChild("EquippedBait") or pData:FindFirstChild("CurrentBait")
                if eq and eq:IsA("StringValue") and eq.Value ~= "" then
                    equippedName = cleanBaitName(eq.Value)
                end
            end
            if not equippedName then
                local attr = pData:GetAttribute("EquippedBait") or pData:GetAttribute("Bait")
                if attr and tostring(attr) ~= "" then
                    equippedName = cleanBaitName(tostring(attr))
                end
            end
        end
    end)
    return equippedName
end

local function isBasicEquippedNow()
    if equipLocked and os.clock() < equipLockUntil then
        return true
    end
    local name = getEquippedBaitName()
    local qty = getEquippedBaitQuantity()
    if name and name:lower() == "basic bait" and qty > 0 then
        equipLocked = true
        equipLockUntil = os.clock() + 60
        return true
    end
    -- nếu đọc được bait KHÁC rõ ràng → mở khóa để equip lại
    if name and name:lower() ~= "basic bait" and qty > 0 then
        equipLocked = false
        return false
    end
    if qty == 0 and name == nil then
        -- không chắc: nếu đang lock thì coi như vẫn ok
        return equipLocked
    end
    return equipLocked
end

local function fireEquipBaitOnce()
    local remote = EquipBaitEvent or evs:FindFirstChild("EquipBait")
    if not remote then return false end
    pcall(function()
        if remote:IsA("RemoteFunction") then
            remote:InvokeServer(BASIC_BAIT_NAME)
        else
            remote:FireServer(BASIC_BAIT_NAME)
        end
    end)
    return true
end

-- CHỈ gọi remote khi chưa basic; SAU KHI gọi → lock, KHÔNG recheck Fire lần 2
function StepAutoEquipBasicBait(forceNotify)
    if not getgenv().AutoEquipBasicBait and not forceNotify then
        return
    end

    -- đã lock / đã basic → tuyệt đối không gửi (tránh toggle tháo)
    if not forceNotify and isBasicEquippedNow() then
        return
    end

    local now = os.clock()
    if not forceNotify and (now - lastEquipBaitClock) < 5 then
        return
    end

    if getBasicBaitCount() <= 0 then
        return
    end

    -- force từ nút UI: cho phép 1 lần, vẫn lock sau đó
    lastEquipBaitClock = now
    if fireEquipBaitOnce() then
        equipLocked = true
        equipLockUntil = now + 60 -- 60s không đụng remote equip
        -- KHÔNG task.delay re-fire (đó là nguyên nhân toggle tháo)
    end
end

    lastBuyBasicClock = now
    local r = evs:FindFirstChild("BuyBait") or evs:FindFirstChild("PurchaseBait") or evs:FindFirstChild("BuyItem")

        pcall(function()
            if r:IsA("RemoteFunction") then
                r:InvokeServer(BASIC_BAIT_NAME, amount)
            else
                r:FireServer(BASIC_BAIT_NAME, amount)
            end
        end)
    elseif evs:FindFirstChild("CraftBait") then
        pcall(function()
            evs.CraftBait:FireServer(BASIC_BAIT_NAME, amount)
        end)
    end
end

local function GetTicketGiverObject()
    local npc = workspace:FindFirstChild("NPC")
    local func = npc and npc:FindFirstChild("Function")
    return func and func:FindFirstChild("Ticket Quest Giver")
end

local function GetActiveTicketQuest()
    local pData = getPlayerData()
    if not pData then return nil end
    local questFolder = pData:FindFirstChild("Quest")
    local mainQ = questFolder and (questFolder:FindFirstChild("Main") or questFolder)
    if not mainQ then return nil end
    for _, n in ipairs({ "Hard Ticket Quest", "Easy Ticket Quest", "Ticket Quest" }) do
        local o = mainQ:FindFirstChild(n)
        if o then return o end
    end
    for _, child in ipairs(mainQ:GetChildren()) do
        if child:IsA("Folder") and child.Name:find("Ticket") then
            return child
        end
    end
    return nil
end

local function IsTicketQuestCompleted(questObj)
    if not questObj then return false end
    local hasNum = false
    for _, progVal in ipairs(questObj:GetChildren()) do
        if progVal:IsA("NumberValue") then
            hasNum = true
            local goal
            local objFolder = questObj:FindFirstChild("Objective")
            local s = objFolder and objFolder:FindFirstChild(progVal.Name)
            if s and s:IsA("StringValue") then
                local parts = string.split(s.Value, ",")
                goal = tonumber(parts[2])
            end
            local maxV = progVal:FindFirstChild("Max")
            if not goal and maxV then goal = maxV.Value end
            if not goal then
                local attr = progVal:GetAttribute("Max")
                if type(attr) == "number" then goal = attr end
            end
            if goal and progVal.Value < goal then
                return false
            end
        end
    end
    return hasNum
end

local function GetTicketQuestStatus()
    local q = GetActiveTicketQuest()
    if not q then return "none" end
    if IsTicketQuestCompleted(q) then return "completed" end
    return "in_progress"
end

local function isDailyTicketLimitText(msg)
    if not msg or msg == "" then return false end
    local m = string.lower(msg)
    return m:find("daily limit", 1, true) and (m:find("20", 1, true) or m:find("come back tomorrow", 1, true))
end

-- Mos-style: "another quest in 20:00" / "15m" / "30s"
local function parseQuestCooldownSeconds(msgText)
    if not msgText or msgText == "" then return nil end
    local lowerMsg = string.lower(msgText)
    if not (string.find(lowerMsg, "another quest in", 1, true)
        or string.find(lowerMsg, "take another quest in", 1, true)
        or string.find(lowerMsg, "wait", 1, true) and string.find(lowerMsg, "quest", 1, true)) then
        return nil
    end
    local minsStr, secsStr = string.match(lowerMsg, "(%d+)%s*:%s*(%d+)")
    if minsStr and secsStr then
        return (tonumber(minsStr) or 0) * 60 + (tonumber(secsStr) or 0)
    end
    local minMatch = string.match(lowerMsg, "(%d+)%s*m")
    local secMatch = string.match(lowerMsg, "(%d+)%s*s")
    if minMatch or secMatch then
        return (tonumber(minMatch) or 0) * 60 + (tonumber(secMatch) or 0)
    end
    return nil
end

local function inspectTicketMessage(msgText)
    if not getgenv().AutoTicketQuest or not msgText or msgText == "" or isTicketLimitReached then return end
    if #msgText > 300 then return end
    if isDailyTicketLimitText(msgText) then
        isTicketLimitReached = true
        getgenv().AutoTicketQuest = false
        pcall(function()
            sg:SetCore("SendNotification", {
                Title = "Ticket",
                Text = "Daily limit reached",
                Duration = 4,
            })
        end)
        return
    end
    if msgText == lastCooldownMsg then return end
    local sec = parseQuestCooldownSeconds(msgText)
    if sec and sec > 0 and sec <= 3600 then
        lastCooldownMsg = msgText
        ticketCooldownUntil = os.clock() + sec
        pcall(function()
            sg:SetCore("SendNotification", {
                Title = "Ticket CD",
                Text = string.format("Wait %dm %ds", math.floor(sec / 60), sec % 60),
                Duration = 3,
            })
        end)
    end
end

-- hook UI text for cooldown / limit
pcall(function()
    local pg = me:FindFirstChild("PlayerGui")
    if not pg then return end
    local function hookLabel(lab)
        if not lab:IsA("TextLabel") and not lab:IsA("TextButton") then return end
        inspectTicketMessage(lab.Text)
        lab:GetPropertyChangedSignal("Text"):Connect(function()
            inspectTicketMessage(lab.Text)
        end)
    end
    for _, d in ipairs(pg:GetDescendants()) do
        hookLabel(d)
    end
    pg.DescendantAdded:Connect(function(obj)
        task.defer(hookLabel, obj)
    end)
end)

local function requestTicketDialogue(giver)
    if not ChooseDialogueOption or not giver then return end
    pcall(function()
        ChooseDialogueOption:FireServer("Ticket Quest Giver", 1, "Quest", { giver })
    end)
    HideTicketDialogueUI()
end

local function acceptTicketQuest(giver)
    if not ChooseDialogueOption or not giver then return end
    local mode = getgenv().TicketQuestMode or "Hard"
    local hard = mode ~= "Easy"
    pcall(function()
        ChooseDialogueOption:FireServer(
            "Ticket Quest Giver",
            hard and 2 or 1,
            hard and "HardAcceptQuest" or "EasyAcceptQuest",
            { giver, "Ticket Quest" }
        )
    end)
    HideTicketDialogueUI()
end

local function stepSmartTicketQuest()
    if not getgenv().AutoTicketQuest or getgenv().AdminIn or isTicketLimitReached then return end
    if not ChooseDialogueOption then
        ChooseDialogueOption = evs:FindFirstChild("ChooseDialogueOption")
        if not ChooseDialogueOption then return end
    end

    local now = os.clock()
    if now < ticketCooldownUntil then
        return -- đang chờ 20p (hoặc số game báo)
    end
    if now - lastTicketAction < TICKET_CD then return end

    -- limit text on open dialogue
    pcall(function()
        local mg = me.PlayerGui:FindFirstChild("MainGui")
        local dialogue = mg and mg:FindFirstChild("Menu") and mg.Menu:FindFirstChild("Dialogue")
        if dialogue and dialogue.Visible then
            for _, lab in ipairs(dialogue:GetDescendants()) do
                if lab:IsA("TextLabel") then
                    inspectTicketMessage(lab.Text)
                end
            end
        end
    end)
    if isTicketLimitReached then return end
    if now < ticketCooldownUntil then return end

    local status = GetTicketQuestStatus()
    if status == "in_progress" then
        return
    end

    local giver = GetTicketGiverObject()
    if not giver then return end

    requestTicketDialogue(giver)
    task.wait(0.45)
    acceptTicketQuest(giver)
    lastTicketAction = os.clock()
end

task.spawn(function()
    while task.wait(1.2) do
        if getgenv().AdminIn then continue end
        stepAutoClaimDailyQuest()
        stepAutoClaimDailyReward()
        if getgenv().AutoTicketQuest then
            pcall(stepSmartTicketQuest)
        end
    end
end)

function unloadHub()
    killAll()
    pcall(function() setWalkOnWater(false) end)
    getgenv().FullBright = false
    getgenv().NoFog = false
    applyLighting()
    clrE()
    for p, d in pairs(getgenv().ESPPlayerList) do
        pcall(function() if d.hl then d.hl:Destroy() end end)
        pcall(function() if d.bill then d.bill:Destroy() end end)
        getgenv().ESPPlayerList[p] = nil
    end
    pcall(function()
        for _, g in ipairs({ game.CoreGui, me.PlayerGui }) do
            if g then
                local u = g:FindFirstChild("BeoHubUI")
                if u then u:Destroy() end
                local f = g:FindFirstChild("BeoFishBtn")
                if f then f:Destroy() end
            end
        end
    end)
    pcall(function()
        sg:SetCore("SendNotification", { Title = "BeoHub", Text = "Unloaded", Duration = 3 })
    end)
end

if getgenv().Water then pcall(setWalkOnWater, true) end
if getgenv().FullBright or getgenv().NoFog then pcall(applyLighting) end
